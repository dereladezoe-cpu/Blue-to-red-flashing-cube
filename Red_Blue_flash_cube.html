// Load the same pinned library version as the class starter.
import * as THREE from "https://cdn.jsdelivr.net/npm/three@0.180.0/build/three.module.js";

// Supplied scene setup. Read each part in the matching slides.
let scene = new THREE.Scene();
let camera = new THREE.PerspectiveCamera(75, 600 / 400, 0.1, 100);
camera.position.z = 3;
let renderer = new THREE.WebGLRenderer();
renderer.setSize(600, 400);
document.getElementById("drawing").appendChild(renderer.domElement); 

// Reuse one geometry and material. Sharing the material shares its colour.
let geometry = new THREE.BoxGeometry(1, 1, 1);
let material = new THREE.MeshBasicMaterial({
  color: "cyan",
  wireframe: true
});

let cube = new THREE.Mesh(geometry, material);
scene.add(cube);

// Click handler: read an input and apply it to the object.
function applyScale() {
  let scale = document.getElementById("scaleInput").valueAsNumber;
  // Supplied validation also catches an empty number field.
  if (Number.isFinite(scale) && scale >= 0.5 && scale <= 1.5) {
    cube.scale.x = scale;
    cube.scale.y = scale;
    cube.scale.z = scale;
    document.getElementById("status").textContent = "Applied scale: " + scale;
  } else {
    document.getElementById("status").textContent = "Enter a scale from 0.5 to 1.5.";
  }
}

function animate() {
  let rotate = document.getElementById("rotateBox").checked;
  if (rotate == true) {
    cube.rotation.x = cube.rotation.x + 0.01;
    cube.rotation.y = cube.rotation.y + 0.01;
  }
  // Keep drawing even while rotation is paused: scale edits still appear.
  renderer.render(scene, camera);
}

document.getElementById("applyButton").addEventListener("click", applyScale);
renderer.setAnimationLoop(animate);
document.getElementById("status").textContent = "Ready. Apply a scale, then untick Rotate.";
